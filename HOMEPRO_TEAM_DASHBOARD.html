// Google Apps Script — วางใน script.google.com
// File → New → Script → วาง code นี้ → Deploy as Web App

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var subject, body;

  if (data.action === 'newTask') {
    subject = '[HomePro] งานใหม่ถูก Assign ให้คุณ: ' + data.taskTitle;
    body = 'สวัสดีครับ/ค่ะ,\n\n'
      + 'มีงานใหม่ถูก Assign ให้คุณจาก Nick SVP:\n\n'
      + '📋 งาน: ' + data.taskTitle + '\n'
      + '📁 Category: ' + data.taskCat + '\n'
      + '🔴 Priority: ' + data.taskPri + '\n'
      + '📅 Due Date: ' + data.taskDue + '\n\n'
      + 'กรุณาตรวจสอบและดำเนินการตามกำหนด\n\n'
      + 'HomePro Team Task Dashboard\n'
      + 'Kitchen & Home Division';
  } else if (data.action === 'dueSoon') {
    subject = '[HomePro] ⚠️ งานใกล้ครบกำหนด พรุ่งนี้: ' + data.taskTitle;
    body = 'แจ้งเตือน: งานของคุณจะครบกำหนดพรุ่งนี้!\n\n'
      + '📋 งาน: ' + data.taskTitle + '\n'
      + '📁 Category: ' + data.taskCat + '\n'
      + '🔴 Priority: ' + data.taskPri + '\n'
      + '📅 Due Date: ' + data.taskDue + '\n\n'
      + 'กรุณาเร่งดำเนินการให้ทันกำหนด\n\n'
      + 'HomePro Team Task Dashboard';
  }

  data.emails.forEach(function(email) {
    GmailApp.sendEmail(email, subject, body);
  });

  return ContentService
    .createTextOutput(JSON.stringify({status:'ok'}))
    .setMimeType(ContentService.MimeType.JSON);
}
